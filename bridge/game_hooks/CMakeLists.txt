cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AetherGameHooks LANGUAGES C CXX)
    include(CTest)
endif()

find_package(Lua 5.4 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(zstd CONFIG REQUIRED)
if(NOT TARGET zstd::libzstd)
    if(TARGET zstd::libzstd_static)
        add_library(zstd::libzstd ALIAS zstd::libzstd_static)
    else()
        add_library(zstd::libzstd ALIAS zstd::libzstd_shared)
    endif()
endif()
find_package(Boost CONFIG REQUIRED COMPONENTS locale)
add_library(aether_game_hooks STATIC src/AetherGameHooks.cpp)
target_compile_features(aether_game_hooks PUBLIC cxx_std_17)
target_include_directories(aether_game_hooks PUBLIC include PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(aether_game_hooks PRIVATE ${LUA_LIBRARIES} ZLIB::ZLIB zstd::libzstd Boost::locale)
set_target_properties(aether_game_hooks PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(aether_lua_probe EXCLUDE_FROM_ALL tests/GameHooksProbe.cpp)
target_link_libraries(aether_lua_probe PRIVATE aether_game_hooks)

if((BUILD_TESTING OR ENABLE_TESTS) AND NOT (IOS OR ANDROID OR WEB OR EMSCRIPTEN))
    find_package(Catch2 3 CONFIG REQUIRED)
    add_executable(aether_game_hooks_tests tests/GameHooksTests.cpp)
    target_link_libraries(aether_game_hooks_tests PRIVATE aether_game_hooks Catch2::Catch2WithMain)
    add_test(NAME aether_game_hooks_tests COMMAND aether_game_hooks_tests)
endif()
