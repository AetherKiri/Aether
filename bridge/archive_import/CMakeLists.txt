# Use the same patched libarchive as the rest of Aether, including the static iOS build.
find_package(LibArchive REQUIRED)
find_package(unofficial-brotli CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(BZip2 REQUIRED)
find_package(LibLZMA REQUIRED)
find_package(lz4 CONFIG REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(Iconv REQUIRED)
add_library(archive_static INTERFACE)
target_link_libraries(archive_static INTERFACE LibArchive::LibArchive
    BZip2::BZip2 LibLZMA::LibLZMA lz4::lz4 zstd::libzstd_static ZLIB::ZLIB Iconv::Iconv)
if(ANDROID)
    find_package(OpenSSL REQUIRED)
    target_link_libraries(archive_static INTERFACE OpenSSL::Crypto)
endif()
add_library(brotlidec INTERFACE)
target_link_libraries(brotlidec INTERFACE unofficial::brotli::brotlidec)
add_subdirectory("${CMAKE_SOURCE_DIR}/packages/AetherUnzip/core" aetherunzip)
# Android shared storage and some external volumes do not support hard links.
# Use the same exclusive publication for decoded entries and the final layout.
set(core_extract "${CMAKE_SOURCE_DIR}/packages/AetherUnzip/core/src/extract.c")
set(fixed_extract "${CMAKE_CURRENT_BINARY_DIR}/portable/extract.c")
set(publish_patch "${CMAKE_CURRENT_SOURCE_DIR}/portable-publish.patch")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${publish_patch}")
configure_file("${core_extract}" "${fixed_extract}" COPYONLY)
find_program(AETHER_PATCH_PROGRAM patch REQUIRED)
execute_process(COMMAND "${AETHER_PATCH_PROGRAM}" -s -F 0 "${fixed_extract}" -i "${publish_patch}"
    COMMAND_ERROR_IS_FATAL ANY)
get_target_property(core_sources aetherunzip SOURCES)
list(REMOVE_ITEM core_sources src/extract.c)
set_property(TARGET aetherunzip PROPERTY SOURCES "${core_sources};${fixed_extract}")
target_include_directories(aetherunzip PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(aether_archive_import STATIC archive_import.cpp archive_plan.cpp)
set_target_properties(aether_archive_import PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(aether_archive_import PUBLIC cxx_std_17)
target_include_directories(aether_archive_import PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(aether_archive_import PUBLIC aetherunzip)
