function(aether_native_host_use_provider target id)
    if(TARGET ${target})
        target_compile_definitions(aether_native_host PRIVATE "AETHER_NATIVE_HOST_WITH_${id}")
        target_link_libraries(aether_native_host PRIVATE ${target})
    endif()
endfunction()
