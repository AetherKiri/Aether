cmake_minimum_required(VERSION 3.24)
project(aether_renpy_probe LANGUAGES CXX)
include(FetchContent)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_TARGET template_debug CACHE STRING "" FORCE)
FetchContent_Declare(godot_cpp
    URL https://codeload.github.com/godotengine/godot-cpp/tar.gz/10.0.0-stable
    URL_HASH SHA256=ebbf64f0f8f8b8b66bc4bc4d3797db97f95b19cf0dfb5b3f8cf56683d9df7db4)
FetchContent_MakeAvailable(godot_cpp)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(renpy_probe SHARED register_types.cpp "${REPO}/bridge/godot_extension/src/renpy_native_player.cpp")
target_compile_features(renpy_probe PRIVATE cxx_std_17)
target_link_libraries(renpy_probe PRIVATE godot-cpp)
target_include_directories(renpy_probe PRIVATE "${REPO}/bridge/godot_extension/src")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(renpy_probe PRIVATE "${REPO}/bridge/godot_extension/src/renpy_native_linux.cpp")
    target_link_libraries(renpy_probe PRIVATE dl)
    add_library(aether_renpy_sdl_hook SHARED "${REPO}/bridge/renpy_runtime/sdl_dynamic_hook.cpp")
    target_link_libraries(aether_renpy_sdl_hook PRIVATE dl)
endif()
