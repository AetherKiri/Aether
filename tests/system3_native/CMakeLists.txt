cmake_minimum_required(VERSION 3.28)
project(AetherSystem3NativeProbe LANGUAGES C CXX)
get_filename_component(AETHER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(AETHERKIRI_SYSTEM3_DIR "${AETHER_ROOT}/packages/system3")
add_subdirectory("${AETHER_ROOT}/bridge/system3_runtime" "${CMAKE_CURRENT_BINARY_DIR}/runtime")
find_package(nlohmann_json CONFIG REQUIRED)
find_package(SDL2_image CONFIG REQUIRED)
add_executable(aether_system3_native_probe probe.cpp)
target_compile_features(aether_system3_native_probe PRIVATE cxx_std_17)
target_include_directories(aether_system3_native_probe PRIVATE "${AETHER_ROOT}/abi/include" "${AETHERKIRI_SYSTEM3_DIR}/src")
target_link_libraries(aether_system3_native_probe PRIVATE aether_system3_runtime nlohmann_json::nlohmann_json
    $<IF:$<TARGET_EXISTS:SDL2_image::SDL2_image>,SDL2_image::SDL2_image,SDL2_image::SDL2_image-static>)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/contracts.cmake")
