cmake_minimum_required(VERSION 3.28)
if(NOT DEFINED AETHERKIRI_XSYSTEM4_DIR)
    get_filename_component(AETHERKIRI_XSYSTEM4_DIR "${CMAKE_CURRENT_LIST_DIR}/../../packages/xsystem4" ABSOLUTE)
endif()
if(NOT EXISTS "${AETHERKIRI_XSYSTEM4_DIR}/src/embedded.c")
    message(FATAL_ERROR "Initialize packages/xsystem4 and its libsys4 submodule")
endif()

find_package(SDL2 CONFIG REQUIRED)
find_package(Freetype REQUIRED)
find_package(SndFile CONFIG REQUIRED)
find_package(cglm CONFIG REQUIRED)
find_package(unofficial-libffi CONFIG REQUIRED)
find_package(uchardet CONFIG REQUIRED)
find_package(libjpeg-turbo CONFIG REQUIRED)
find_package(FFMPEG REQUIRED)
if(APPLE)
    list(REMOVE_ITEM FFMPEG_LIBRARIES "-pthread")
endif()
find_package(Threads REQUIRED)
find_package(FLEX REQUIRED)
find_package(BISON REQUIRED)
set(CMAKE_DISABLE_FIND_PACKAGE_libdeflate TRUE)
add_subdirectory("${AETHERKIRI_XSYSTEM4_DIR}/subprojects/libsys4" "${CMAKE_CURRENT_BINARY_DIR}/libsys4")

# libsys4 assumes a static TurboJPEG target, but Linux triplets export the
# shared target. Adapt both the archive's direct and transitive dependencies.
foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(value sys4 ${property})
    string(REPLACE "libjpeg-turbo::turbojpeg-static"
        "$<IF:$<TARGET_EXISTS:libjpeg-turbo::turbojpeg>,libjpeg-turbo::turbojpeg,libjpeg-turbo::turbojpeg-static>"
        value "${value}")
    set_property(TARGET sys4 PROPERTY ${property} "${value}")
endforeach()

set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}")
file(READ "${AETHERKIRI_XSYSTEM4_DIR}/subprojects/libsys4/src/system.c" system_source)
if(NOT system_source MATCHES "exit\\(code\\);")
    message(FATAL_ERROR "libsys4 exit boundary changed")
endif()
string(REPLACE "exit(code);" "xsystem4_host_exit(code);" system_source "${system_source}")
file(WRITE "${generated}/system.c" "#include \"embedded.h\"\n${system_source}")
get_target_property(sys4_sources sys4 SOURCES)
list(FILTER sys4_sources EXCLUDE REGEX "(^|/)src/system\\.c$")
set_property(TARGET sys4 PROPERTY SOURCES "${sys4_sources};${generated}/system.c")
target_include_directories(sys4 PRIVATE "${AETHERKIRI_XSYSTEM4_DIR}/include")
# Keep the pinned upstream dependency unchanged; adapt its AFA reader in the
# build directory, as with the embedded exit boundary above.
set(afa_path "${AETHERKIRI_XSYSTEM4_DIR}/subprojects/libsys4/src/afa.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${afa_path}")
file(READ "${afa_path}" afa_source)
set(afa_decompress "zlib_decompress_exact(table, ar->uncompressed_size, buf, ar->compressed_size)")
set(afa_size "ar->compressed_size = LittleEndian_getDW((uint8_t*)buf, 32) - 16;")
string(FIND "${afa_source}" "${afa_decompress}" afa_decompress_pos)
string(FIND "${afa_source}" "${afa_size}" afa_size_pos)
if(afa_decompress_pos LESS 0 OR afa_size_pos LESS 0)
    message(FATAL_ERROR "libsys4 AFA index boundary changed")
endif()
string(REPLACE "${afa_decompress}" "aether_afa_decompress_index(f, ar, table, &buf)" afa_source "${afa_source}")
string(REPLACE "${afa_size}" [=[
	uint32_t info_size = LittleEndian_getDW((uint8_t*)buf, 32);
	if (info_size < 16 || ar->data_start < 44 ||
	    ar->data_start > ar->file_size || info_size - 16 > ar->data_start - 44) {
		*error = ARCHIVE_BAD_ARCHIVE_ERROR;
		return false;
	}
	ar->compressed_size = info_size - 16;]=] afa_source "${afa_source}")
file(WRITE "${generated}/afa.c" "#include \"afa_index_compat.h\"\n${afa_source}")
get_target_property(sys4_sources sys4 SOURCES)
list(FILTER sys4_sources EXCLUDE REGEX "(^|/)src/afa\\.c$")
set_property(TARGET sys4 PROPERTY SOURCES "${sys4_sources};${generated}/afa.c")
target_include_directories(sys4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/src")
target_link_libraries(sys4 PRIVATE $<IF:$<TARGET_EXISTS:SDL2::SDL2>,SDL2::SDL2,SDL2::SDL2-static>)
set_target_properties(sys4 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
    find_library(core_foundation CoreFoundation REQUIRED)
    # The GDExtension's mixed Swift/C++ link needs driver-neutral framework
    # paths, rather than libsys4's standalone clang flag spelling.
    foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
        get_target_property(value sys4 ${property})
        string(REPLACE "-framework CoreFoundation" "${core_foundation}" value "${value}")
        set_property(TARGET sys4 PROPERTY ${property} "${value}")
    endforeach()
endif()

file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${AETHERKIRI_XSYSTEM4_DIR}/src/*.c")
list(FILTER sources EXCLUDE REGEX "/(system4|movie_plmpeg|debugger|debugger_scm|debug-ffi)\\.c$")
list(FILTER sources EXCLUDE REGEX "/src/debug[^/]*\\.c$")
set(VCS_TAG "aether-native")
configure_file("${AETHERKIRI_XSYSTEM4_DIR}/src/version.h.in" "${generated}/version.h")
file(GLOB_RECURSE shaders CONFIGURE_DEPENDS RELATIVE "${AETHERKIRI_XSYSTEM4_DIR}" "${AETHERKIRI_XSYSTEM4_DIR}/shaders/*.glsl")
set(resources "#pragma once\nstruct Xsystem4Shader { const char *path; const char *source; };\nstatic const Xsystem4Shader xsystem4_shaders[] = {\n")
foreach(shader IN LISTS shaders)
    file(READ "${AETHERKIRI_XSYSTEM4_DIR}/${shader}" source)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AETHERKIRI_XSYSTEM4_DIR}/${shader}")
    string(APPEND resources "{\"${shader}\", R\"XS4SHADER(${source})XS4SHADER\"},\n")
endforeach()
file(WRITE "${generated}/shader_resources.h" "${resources}};\n")

add_library(aether_xsystem4_runtime STATIC src/xsystem4_runtime_provider.cpp ${sources})
target_compile_features(aether_xsystem4_runtime PRIVATE cxx_std_17)
target_compile_definitions(aether_xsystem4_runtime PRIVATE _DEFAULT_SOURCE XSYSTEM4_EMBEDDED SDL_MAIN_HANDLED XSYS4_DATA_DIR="" VCS_TAG="aether")
if(APPLE)
    target_compile_definitions(aether_xsystem4_runtime PRIVATE GL_SILENCE_DEPRECATION)
endif()
target_include_directories(aether_xsystem4_runtime PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include" PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/../../abi/include" "${AETHERKIRI_XSYSTEM4_DIR}/include" "${generated}" ${FFMPEG_INCLUDE_DIRS})
target_link_libraries(aether_xsystem4_runtime PRIVATE sys4
    $<IF:$<TARGET_EXISTS:SDL2::SDL2>,SDL2::SDL2,SDL2::SDL2-static>
    Freetype::Freetype SndFile::sndfile cglm::cglm unofficial::libffi::libffi
    Threads::Threads uchardet::libuchardet ${FFMPEG_LIBRARIES})
set_target_properties(aether_xsystem4_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
    target_sources(aether_xsystem4_runtime PRIVATE src/offscreen_context_apple.mm)
    set_source_files_properties(src/offscreen_context_apple.mm PROPERTIES LANGUAGE CXX COMPILE_FLAGS "-x objective-c++ -fobjc-arc -Wno-deprecated-declarations")
    if(IOS)
        target_compile_definitions(aether_xsystem4_runtime PRIVATE USE_GLES)
        find_library(gl_framework OpenGLES REQUIRED)
    else()
        find_library(gl_framework OpenGL REQUIRED)
    endif()
    target_link_libraries(aether_xsystem4_runtime PRIVATE "${gl_framework}" iconv)
elseif(ANDROID OR UNIX)
    target_sources(aether_xsystem4_runtime PRIVATE src/offscreen_context_egl.cpp)
    target_compile_definitions(aether_xsystem4_runtime PRIVATE USE_GLES)
    if(ANDROID)
        target_link_libraries(aether_xsystem4_runtime PRIVATE EGL GLESv3 log)
    else()
        find_library(egl_library EGL REQUIRED)
        find_library(gles_library GLESv2 REQUIRED)
        target_link_libraries(aether_xsystem4_runtime PRIVATE "${egl_library}" "${gles_library}")
    endif()
else()
    message(FATAL_ERROR "xsystem4 native host requires Apple, Linux or Android")
endif()

if(TARGET engine_api)
    target_link_libraries(aether_xsystem4_runtime PRIVATE engine_api)
endif()
